import { getCollection } from 'astro:content';
import { releaseSlugFromVersion } from '../../../lib/releaseEntry.js';
import HtmlContent from '../../../components/HtmlContent.astro';

export const releases = (await getCollection('releases')).sort(
  (a, b) => b.data.date.valueOf() - a.data.date.valueOf()
);

All notable changes to Peppy will be documented on this page.

Subscribe to the [Atom feed](/changelog.xml) for updates.

{releases.map((release, index) => (
  <details open={index === 0}>
    <summary>
      <strong><a href={`/releases/${releaseSlugFromVersion(release.data.version)}/`}>v{release.data.version} (Alpha)</a></strong>: <em>{release.data.description}</em>
    </summary>
    <HtmlContent html={release.body ?? ''} />
  </details>
))}

import { getCollection, render } from 'astro:content';

export const releases = (await getCollection('releases')).sort(
  (a, b) => b.data.date.valueOf() - a.data.date.valueOf()
);

export const renderedReleases = await Promise.all(
  releases.map(async (release) => {
    const { Content } = await render(release);
    return { release, Content };
  })
);

All notable changes to PeppyOS will be documented on this page.

Subscribe to the [Atom feed](/changelog.xml) for updates.

{renderedReleases.map(({ release, Content }, index) => (
  <details open={index === 0}>
    <summary>
      <strong>v{release.data.version} (Alpha)</strong> — <em>{release.data.description}</em>
    </summary>
    <Content />
  </details>
))}

class ActionHandle:
    @classmethod
    async def fire_goal(cls, node_runner: peppylib.NodeRunner, target: peppylib.ProducerRef, request: GoalRequest, timeout: float, feedback_bound: Optional[int] = None) -> Self:
        """Fires this goal at `target`, a member of the slot's bound set.

        A target outside the set fails before anything reaches the wire. The
        handle retains the target: feedback, result, and cancel stay pinned
        to it. The handle keeps every feedback message until it is read, up to
        `feedback_bound` unread messages (None takes the default of peppylib,
        1024); a caller that falls behind by more cancels the goal and reads
        peppylib.FeedbackOverflowError. `refusal` holds the failure of a
        refused goal; a kind that this member does not declare raises
        ValueError.
        This slot declares cardinality `one`: `target` is its sole bound
        producer, returned by `bound_producer()`; the explicit parameter
        keeps call sites uniform across cardinalities.
        """
        node_runner.ensure_target_bound(LINK_ID, target)
        capnp_msg = _arm_control_move_arm_fire_goal_message_capnp().ArmControlMoveArmFireGoalMessage.new_message()
        capnp_msg.armId = request.arm_id
        if len(request.desired_position) != 3:
            raise ValueError("invalid fixed list length for field 'desired_position': expected 3, got " + str(len(request.desired_position)))
        capnp_msg.desiredPosition = request.desired_position
        user_goal_payload = capnp_msg.to_bytes()
        action_handle = await peppylib.ActionMessenger.send_goal(
            node_runner.messenger(),
            node_runner.bound_core_node(),
            node_runner.bound_instance_id(),
            peppylib.SenderTarget.contract("arm_control", "v1"),
            TARGET_ACTION_NAME,
            target,
            user_goal_payload,
            feedback_bound,
            timeout,
        )
        handle = cls()
        handle._messenger = node_runner.messenger()
        handle._inner = action_handle
        handle.accepted = action_handle.accepted
        handle.refusal = None if action_handle.accepted else Failure(action_handle.refusal_kind, action_handle.refusal_message)
        handle.data = _deserialize_goal_response(action_handle.goal_reply_body) if action_handle.accepted else None
        return handle

    async def cancel_goal(self, timeout: float) -> CancelResponse:
        reply = await peppylib.ActionMessenger.cancel_goal(
            self._messenger,
            self._inner,
            timeout,
        )
        return CancelResponse(core_node=reply.core_node, instance_id=reply.instance_id, state=CancelState(reply.state))

    async def on_next_feedback_message(self) -> FeedbackMessage:
        """Receive the next feedback message for this goal.

        Raises RuntimeError when the producer closed the stream cleanly
        (end-of-stream sentinel), ConnectionError when the producer
        instance disappeared without closing it (process killed, crashed),
        in which case get_result resolves to ResultStatus.ABANDONED, and
        peppylib.FeedbackOverflowError when this caller fell behind by more
        unread messages than its bound, in which case the handle cancelled
        the goal.
        """
        feedback = await self._inner.on_next_feedback()
        payload = feedback.payload
        return _deserialize_feedback_payload(payload)

    async def get_result(self, timeout: float) -> ResultResponse:
        reply = await peppylib.ActionMessenger.request_result(
            self._messenger,
            self._inner,
            timeout,
        )
        status = ResultStatus(reply.status)
        failure = None
        if status is ResultStatus.FAILED:
            failure = Failure(reply.failure_kind, reply.failure_message)
        data = None
        if status in (ResultStatus.COMPLETED, ResultStatus.CANCELLED):
            data = _deserialize_result_response(reply.body)
        return ResultResponse(core_node=reply.core_node, instance_id=reply.instance_id, status=status, data=data, failure=failure, cancel_reason=reply.cancel_reason)

class FailureKind(str, Enum):
    """A failure kind of this member: a kind of peppy, or a kind that its contract declares."""

    ATTEMPT_FAILED = "attempt_failed"
    "The attempt gave no result"
    TIMEOUT = "timeout"
    "The bound of the provider passed"
    PRECONDITION_FAILED = "precondition_failed"
    "A precondition failed"
    RESOURCE_UNAVAILABLE = "resource_unavailable"
    "A robot, limb or host is gone"
    REJECTED = "rejected"
    "The provider refused the goal, and no other kind says why"


class Failure(peppylib.FailureError):
    """Why this member refused or failed a goal, or gave an error reply.

    A kind that this member declares, and a message for a person. The
    constructor raises ValueError for any other kind, with the message in
    its text, and TypeError for a message that is not a str. A service
    handler raises or returns it to give an error reply of its kind.
    """

    kind: FailureKind
    message: str

    def __init__(self, kind: FailureKind | str, message: str) -> None:
        if not isinstance(message, str):
            raise TypeError(f"the message of a failure of {SERVICE_NAME} is a str, not {type(message).__name__}")
        try:
            parsed = FailureKind(kind)
        except ValueError:
            raise ValueError(f"{kind!r} is not a failure kind of {SERVICE_NAME} (message: {message!r})") from None
        super().__init__(parsed.value, message)
        self.kind = parsed
        self.message = message

    def __eq__(self, other: object) -> bool:
        if not isinstance(other, Failure):
            return NotImplemented
        return self.kind is other.kind and self.message == other.message

    def __hash__(self) -> int:
        return hash((self.kind, self.message))

    def __repr__(self) -> str:
        return f"Failure({self.kind!r}, {self.message!r})"

    def __str__(self) -> str:
        return f"{self.kind.value}: {self.message}"



def bound_producer(node_runner: peppylib.NodeRunner) -> peppylib.ProducerRef:

    The binding is fixed when the node starts (no live discovery; a
    producer disconnecting never rebinds it) and shared by every
    generated module referencing this slot. This slot declares
    cardinality `one`: launch validation resolved exactly one producer,
    so the accessor is singular and infallible.
    """
    return node_runner.bound_producer("bare_link")

    """Polls this service on `target`, a member of the slot's bound set.

    A target outside the set fails before anything reaches the wire. An
    error reply of the provider raises this member's `Failure`, a
    peppylib.FailureError; a kind that this member does not declare raises
    ValueError.
    This slot declares cardinality `one`: `target` is its sole bound

    """
    node_runner.ensure_target_bound(LINK_ID, target)
    request_payload = b""

            node_runner.bound_core_node(),

class FailureKind(str, Enum):
    """A failure kind of this member: a kind of peppy, or a kind that its contract declares."""

    ATTEMPT_FAILED = "attempt_failed"
    "The attempt gave no result"
    TIMEOUT = "timeout"
    "The bound of the provider passed"
    PRECONDITION_FAILED = "precondition_failed"
    "A precondition failed"
    RESOURCE_UNAVAILABLE = "resource_unavailable"
    "A robot, limb or host is gone"
    REJECTED = "rejected"
    "The provider refused the goal, and no other kind says why"


class Failure(peppylib.FailureError):
    """Why this member refused or failed a goal, or gave an error reply.

    A kind that this member declares, and a message for a person. The
    constructor raises ValueError for any other kind, with the message in
    its text, and TypeError for a message that is not a str. A service
    handler raises or returns it to give an error reply of its kind.
    """

    kind: FailureKind
    message: str

    def __init__(self, kind: FailureKind | str, message: str) -> None:
        if not isinstance(message, str):
            raise TypeError(f"the message of a failure of {TARGET_ACTION_NAME} is a str, not {type(message).__name__}")
        try:
            parsed = FailureKind(kind)
        except ValueError:
            raise ValueError(f"{kind!r} is not a failure kind of {TARGET_ACTION_NAME} (message: {message!r})") from None
        super().__init__(parsed.value, message)
        self.kind = parsed
        self.message = message

    def __eq__(self, other: object) -> bool:
        if not isinstance(other, Failure):
            return NotImplemented
        return self.kind is other.kind and self.message == other.message

    def __hash__(self) -> int:
        return hash((self.kind, self.message))

    def __repr__(self) -> str:
        return f"Failure({self.kind!r}, {self.message!r})"

    def __str__(self) -> str:
        return f"{self.kind.value}: {self.message}"

class CancelState(IntEnum):
    SIGNALLED = 0
    ALREADY_TERMINAL = 1
    UNKNOWN = 2

@dataclass
class CancelResponse:
    core_node: str
    instance_id: str
    state: CancelState

class ResultStatus(IntEnum):
    COMPLETED = 0
    CANCELLED = 1
    ABANDONED = 2
    EXPIRED = 3
    FAILED = 4

@dataclass
class ResultResponse:
    core_node: str
    instance_id: str
    status: ResultStatus
    failure: Optional[Failure]
    cancel_reason: Optional[str]


def bound_producer(node_runner: peppylib.NodeRunner) -> peppylib.ProducerRef:
    """The producer bound to this module's slot.

    The binding is fixed when the node starts (no live discovery; a
    producer disconnecting never rebinds it) and shared by every
    generated module referencing this slot. This slot declares
    cardinality `one`: launch validation resolved exactly one producer,
    so the accessor is singular and infallible.
    """
    return node_runner.bound_producer("bare_link")


class ActionHandle:
    @classmethod
    async def fire_goal(cls, node_runner: peppylib.NodeRunner, target: peppylib.ProducerRef, timeout: float, feedback_bound: Optional[int] = None) -> Self:
        """Fires this goal at `target`, a member of the slot's bound set.

        A target outside the set fails before anything reaches the wire. The
        handle retains the target: feedback, result, and cancel stay pinned
        to it. The handle keeps every feedback message until it is read, up to
        `feedback_bound` unread messages (None takes the default of peppylib,
        1024); a caller that falls behind by more cancels the goal and reads
        peppylib.FeedbackOverflowError. `refusal` holds the failure of a
        refused goal; a kind that this member does not declare raises
        ValueError.
        This slot declares cardinality `one`: `target` is its sole bound
        producer, returned by `bound_producer()`; the explicit parameter
        keeps call sites uniform across cardinalities.
        """
        node_runner.ensure_target_bound(LINK_ID, target)
        user_goal_payload = b""
        action_handle = await peppylib.ActionMessenger.send_goal(
            node_runner.messenger(),
            node_runner.bound_core_node(),
            node_runner.bound_instance_id(),
            peppylib.SenderTarget.node(TARGET_NODE_NAME, "v1"),
            TARGET_ACTION_NAME,
            target,
            user_goal_payload,
            feedback_bound,
            timeout,
        )
        handle = cls()
        handle._messenger = node_runner.messenger()
        handle._inner = action_handle
        handle.accepted = action_handle.accepted
        handle.refusal = None if action_handle.accepted else Failure(action_handle.refusal_kind, action_handle.refusal_message)
        return handle

    async def cancel_goal(self, timeout: float) -> CancelResponse:
        reply = await peppylib.ActionMessenger.cancel_goal(
            self._messenger,
            self._inner,
            timeout,
        )
        return CancelResponse(core_node=reply.core_node, instance_id=reply.instance_id, state=CancelState(reply.state))

    async def get_result(self, timeout: float) -> ResultResponse:
        reply = await peppylib.ActionMessenger.request_result(
            self._messenger,
            self._inner,
            timeout,
        )
        status = ResultStatus(reply.status)
        failure = None
        if status is ResultStatus.FAILED:
            failure = Failure(reply.failure_kind, reply.failure_message)
        return ResultResponse(core_node=reply.core_node, instance_id=reply.instance_id, status=status, failure=failure, cancel_reason=reply.cancel_reason)

    """What `start` returns: awaitable (`harness = await start(...)`) and async context manager (`async with start(...) as harness:`, shutdown runs even when the test body fails)."""

        self._coro = coro

        self._harness = await self._coro
        return self._harness

    """Boots the full fixture: ephemeral router + started mocks + seeded
    StandaloneConfig + the node running in-process behind the readiness
    barriers. `setup` is the node's real entry point (the exact
    `async def setup(params, node_runner)` shape `NodeBuilder().run`
    takes), spawned only after every readiness barrier passed.

        harness = await start(setup)   # then: await harness.shutdown()
        async with start(setup) as harness: ...   # shutdown on exit

    `parameters` is the typed `peppygen.parameters.Parameters` override
    (`None` uses the schema defaults); `instance_id` overrides the unique
    generated one; `node_dir` points at the directory holding the node's

            peppylib.testing.MOCK_CLOCK_INSTANCE_ID,

            peppylib.StandaloneConfig()
            .with_messaging(router.host, router.port)
            .with_instance_id(instance_id)
            .with_clock(clock.binding())
        )

        standalone = standalone.with_bound_producer("bare_link", _dep_bare_link.MOCK_CORE_NODE, _dep_bare_link.MOCK_INSTANCE_ID)

            os.path.join(node_dir, _NODE_CONFIG_FILE),

            publisher_readiness,
            setup,
            parameters=_hydrate_parameters(parameters),
            service_readiness=service_readiness,
        )

            core=core,
            mocks=Mocks(

                pairings=PairingMocks(),
                observed=ObservedMocks(),

            emitted=Emitted(),

            session=session,
            router=router,
            instance_id=instance_id,
        )

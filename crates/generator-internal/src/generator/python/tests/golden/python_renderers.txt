ACTION_NAME = "move_arm"

@dataclass
class GoalRequestData:
    arm_id: int
    desired_position: list[int]

@dataclass
class GoalRequest:
    instance_id: str
    core_node: str
    data: GoalRequestData

@dataclass
class GoalResponse:
    accepted: bool

class GoalDecision:
    def __init__(self, accepted: bool, reason=None, response=None):
        self.accepted = accepted
        self.reason = reason
        self.response = response
    @staticmethod
    def accept(response):
        if response is None:
            raise ValueError("GoalDecision.accept requires the declared GoalResponse")
        return GoalDecision(True, response=response)
    @staticmethod
    def reject(reason=None, response=None):
        return GoalDecision(False, reason=reason, response=response)


def _deserialize_goal_request(payload: bytes) -> GoalRequestData:
    with _move_arm_goal_request_message_capnp().MoveArmGoalRequestMessage.from_bytes(payload) as capnp_msg:
        arm_id_0 = capnp_msg.armId
        desired_position_1 = list(capnp_msg.desiredPosition)
        if len(desired_position_1) != 3:
            raise ValueError("invalid fixed list length for field 'desired_position': expected 3, got " + str(len(desired_position_1)))
        return GoalRequestData(arm_id=arm_id_0, desired_position=desired_position_1)

class ActionHandle:
    @classmethod
    async def expose(cls, node_runner: peppylib.NodeRunner) -> Self:
        inner = await peppylib.ConcurrentAction.expose(
            node_runner.messenger(),
            node_runner.bound_core_node(),
            node_runner.bound_instance_id(),
            peppylib.SenderTarget.node(node_runner.node_name(), node_runner.node_tag()),
            ACTION_NAME,
            True,
        )
        handle = cls()
        handle._inner = inner
        return handle

    async def handle_goal_next_request(self, handler: Callable[[GoalRequest], GoalDecision]) -> "GoalContext | None":
        while True:
            pending = await self._inner.recv_next_goal()
            if pending is None:
                return None  # goal stream closed (node shutting down)
            try:
                request_data = _deserialize_goal_request(pending.request_bytes)
            except (ValueError, capnp.KjException) as error:
                await pending.reject(f"goal request does not decode: {error}", b"")
                continue
            request = GoalRequest(instance_id=pending.instance_id, core_node=pending.core_node, data=request_data)
            decision = handler(request)
            if hasattr(decision, "__await__"):
                decision = await decision
            response = decision.response
            if response is not None:
                capnp_msg = _move_arm_goal_response_message_capnp().MoveArmGoalResponseMessage.new_message()
                capnp_msg.accepted = response.accepted
                response_bytes = capnp_msg.to_bytes()
            else:
                response_bytes = b""
            if decision.accepted:
                ctx = await pending.accept(response_bytes)
                return GoalContext(ctx, request)
            await pending.reject(decision.reason, response_bytes)

class GoalContext:
    def __init__(self, inner, request: GoalRequest):
        self._inner = inner
        self._request = request

    def request(self) -> GoalRequest:
        return self._request
    def goal_id(self) -> str:
        return self._inner.goal_id
    async def cancel_signal(self) -> None:
        await self._inner.cancel_signal()
    def is_cancelled(self) -> bool:
        return self._inner.is_cancelled()
    async def publish_feedback(self, new_position: list[int]) -> None:
        capnp_msg = _move_arm_feedback_message_capnp().MoveArmFeedbackMessage.new_message()
        if len(new_position) != 3:
            raise ValueError("invalid fixed list length for field 'new_position': expected 3, got " + str(len(new_position)))
        capnp_msg.newPosition = new_position
        payload = capnp_msg.to_bytes()
        await self._inner.publish_feedback(payload)
    async def complete(self, success: bool, error_msg: Optional[str]) -> None:
        capnp_msg = _move_arm_result_response_message_capnp().MoveArmResultResponseMessage.new_message()
        capnp_msg.success = success
        if error_msg is not None:
            capnp_msg.errorMsg = error_msg
        payload = capnp_msg.to_bytes()
        await self._inner.complete(payload)
    async def complete_cancelled(self, success: bool, error_msg: Optional[str]) -> None:
        capnp_msg = _move_arm_result_response_message_capnp().MoveArmResultResponseMessage.new_message()
        capnp_msg.success = success
        if error_msg is not None:

class ActionHandle:
    @classmethod
    async def fire_goal(cls, node_runner: peppylib.NodeRunner, target: peppylib.ProducerRef, timeout: float, feedback_qos: peppylib.QoSProfile) -> Self:
        """Fires this goal at `target`, a member of the slot's bound set.

        A target outside the set fails before anything reaches the wire. The
        handle retains the target: feedback, result, and cancel stay pinned
        to it.
        This slot declares cardinality `one`: `target` is its sole bound
        producer, returned by `bound_producer()`; the explicit parameter
        keeps call sites uniform across cardinalities.
        """
        node_runner.ensure_target_bound(LINK_ID, target)
        user_goal_payload = b""
        action_handle = await peppylib.ActionMessenger.send_goal(
            node_runner.messenger(),
            node_runner.bound_core_node(),
            node_runner.bound_instance_id(),
            peppylib.SenderTarget.node(TARGET_NODE_NAME, "v1"),
            TARGET_ACTION_NAME,
            target,
            user_goal_payload,
            feedback_qos,
            timeout,
        )
        handle = cls()
        handle._messenger = node_runner.messenger()
        handle._inner = action_handle
        handle.accepted = action_handle.accepted
        handle.reason = action_handle.reason
        return handle

    async def cancel_goal(self, timeout: float) -> CancelResponse:
        reply = await peppylib.ActionMessenger.cancel_goal(
            self._messenger,
            self._inner,
            timeout,
        )
        return CancelResponse(core_node=reply.core_node, instance_id=reply.instance_id, state=CancelState(reply.state))

    async def get_result(self, timeout: float) -> ResultResponse:
        reply = await peppylib.ActionMessenger.request_result(
            self._messenger,
            self._inner,
            timeout,
        )
        status = ResultStatus(reply.status)
        return ResultResponse(core_node=reply.core_node, instance_id=reply.instance_id, status=status)
